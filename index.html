// 本地存儲或使用預設

let questionBank = JSON.parse(localStorage.getItem('yoga_quiz_db')) || defaultQuestionBank;


let currentQuiz = [];

let currentIndex = 0;

let userAnswers = [];


function showCard(id) {

    document.querySelectorAll('.card').forEach(c => c.classList.remove('active'));

    document.getElementById(id).classList.add('active');

}


function showHome() { showCard('home-card'); }

function showEditor() { showCard('editor-card'); }


function startQuiz() {

    // 隨機抽 5 題（或全部題目）

    currentQuiz = [...questionBank].sort(() => 0.5 - Math.random()).slice(0, 5);

    currentIndex = 0;

    userAnswers = [];

    showCard('quiz-card');

    renderQuestion();

}


function renderQuestion() {

    let q = currentQuiz[currentIndex];

    document.getElementById('progress').style.width = `${((currentIndex + 1) / currentQuiz.length) * 100}%`;

    document.getElementById('category-badge').innerText = `[${q.category}] 第 ${currentIndex + 1} / ${currentQuiz.length} 題`;

    document.getElementById('question-text').innerText = q.question;

    

    let imgEl = document.getElementById('question-img');

    if (q.img) {

        imgEl.src = q.img;

        imgEl.style.display = 'block';

    } else {

        imgEl.style.display = 'none';

    }


    let optsContainer = document.getElementById('options-container');

    let inputEl = document.getElementById('short-answer-input');

    optsContainer.innerHTML = '';


    if (q.type === 'single') {

        inputEl.style.display = 'none';

        optsContainer.style.display = 'block';

        q.options.forEach((opt, i) => {

            let letter = String.fromCharCode(65 + i);

            let btn = document.createElement('button');

            btn.className = 'option-btn';

            btn.innerText = `${letter}. ${opt}`;

            btn.onclick = () => {

                document.querySelectorAll('.option-btn').forEach(b => b.classList.remove('selected'));

                btn.classList.add('selected');

                userAnswers[currentIndex] = letter;

            };

            optsContainer.appendChild(btn);

        });

    } else {

        optsContainer.style.display = 'none';

        inputEl.style.display = 'block';

        inputEl.value = '';

        inputEl.oninput = (e) => { userAnswers[currentIndex] = e.target.value; };

    }

}


function nextQuestion() {

    if (!userAnswers[currentIndex]) {

        alert("請先回答或填寫選項喔！");

        return;

    }

    currentIndex++;

    if (currentIndex < currentQuiz.length) {

        renderQuestion();

    } else {

        calculateResult();

    }

}


function calculateResult() {

    let score = 0;

    let categoryStats = {};


    currentQuiz.forEach((q, i) => {

        if (!categoryStats[q.category]) categoryStats[q.category] = { correct: 0, total: 0 };

        categoryStats[q.category].total++;


        let isCorrect = false;

        if (q.type === 'single') {

            isCorrect = userAnswers[i] === q.answer;

        } else {

            // 簡答題關鍵字包含匹配

            isCorrect = userAnswers[i] && q.answer.split('、').every(k => userAnswers[i].includes(k.trim()));

        }


        if (isCorrect) {

            score += (100 / currentQuiz.length);

            categoryStats[q.category].correct++;

        }

    });


    // 顯示分數

    document.getElementById('score-text').innerText = `${Math.round(score)} 分`;


    // 強弱項標籤

    let tagsContainer = document.getElementById('analysis-tags');

    tagsContainer.innerHTML = '';

    for (let cat in categoryStats) {

        let rate = categoryStats[cat].correct / categoryStats[cat].total;

        let tag = document.createElement('span');

        tag.className = `tag ${rate >= 0.6 ? 'tag-strong' : 'tag-weak'}`;

        tag.innerText = `${cat} (${Math.round(rate * 100)}%)`;

        tagsContainer.appendChild(tag);

    }


    // 解析總覽

    let reviewContainer = document.getElementById('review-container');

    reviewContainer.innerHTML = '';

    currentQuiz.forEach((q, i) => {

        let div = document.createElement('div');

        div.className = 'analysis-item';

        div.innerHTML = `

            <strong>Q${i+1}. [${q.category}] ${q.question}</strong><br>

            你的回答：<span style="color:${userAnswers[i] === q.answer ? 'green':'red'}">${userAnswers[i] || '未答'}</span> | 

            正確答案：<b>${q.answer}</b><br>

            <small style="color:#555;">💡 解析：${q.explanation}</small>

        `;

        reviewContainer.appendChild(div);

    });


    showCard('result-card');

}


function toggleTypeEdit() {

    let type = document.getElementById('edit-type').value;

    document.getElementById('edit-options-group').style.display = type === 'single' ? 'block' : 'none';

}


function saveQuestion() {

    let cat = document.getElementById('edit-cat').value;

    let type = document.getElementById('edit-type').value;

    let qText = document.getElementById('edit-q').value;

    let img = document.getElementById('edit-img').value;

    let ans = document.getElementById('edit-ans').value;

    let exp = document.getElementById('edit-exp').value;


    if (!qText || !ans) { alert("請填寫題目與答案！"); return; }


    let opts = [];

    if (type === 'single') {

        document.querySelectorAll('.edit-opt').forEach(input => {

            if (input.value) opts.push(input.value);

        });

    }


    let newQ = {

        id: Date.now(),

        category: cat,

        type: type,

        question: qText,

        img: img,

        options: opts,

        answer: ans.toUpperCase(),

        explanation: exp

    };


    questionBank.push(newQ);

    localStorage.setItem('yoga_quiz_db', JSON.stringify(questionBank));

    alert("題目新增成功！");

    showHome();

}
